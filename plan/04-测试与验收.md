def test_unsymbolizable_not_promoted():
    """
    守门：unsymbolizable 步骤不得虚高为 ok。
    """
    result = normalize("不可解析的 LaTeX: \u005c\u0072\u0061\u006e\u0064\u006f\u006d\u005f\u0063\u006f\u006e\u0073\u0074\u0072\u0075\u0063\u0074")

### 3.4 数值口径唯一定义点

```python
def test_tolerance_single_source():
    """
    守门：TOLERANCE 字面量仅出现在 numeric_recalc.py 与 benchmark/runner.py。
    """
    import ast, pathlib
    forbidden_files = [
        f for f in pathlib.Path("src").rglob("*.py")
        if "numeric_recalc" not in str(f) and "benchmark" not in str(f)
    ]
    for f in forbidden_files:
        tree = ast.parse(f.read_text())
        for node in ast.walk(tree):
            if isinstance(node, ast.Constant) and node.value == 1e-9:
                pytest.fail(f"TOLERANCE 字面量出现在 {f}")
```

### 3.5 坏实现探活

```python
def test_broken_impl_probe():
    """
    守门：故意弄坏被测行为 → 门必须变红。
    """
    class BrokenVerifier:
        def verify(self, step):
            return Verdict("ok", "", None)  # 总是 ok

    results = benchmark.run(BrokenVerifier())
    metrics = compute_metrics(results)
    # 坏实现必须暴露
    assert metrics.false_positive_rate > 0 or metrics.localization_accuracy < 1.0
```

## 五、测试环境

### 5.1 双解释器

- 3.8（底线）+ 3.12（最新）
- 全量测试在两个版本上各跑一次

### 5.2 编码

- 所有测试文件 UTF-8
- 用 `TextIOWrapper(BytesIO(), encoding="cp1252", errors="strict")` 复现 CI 编码环境

### 5.3 确定性

- 同输入同结论（双跑 IR 逐字节一致）
- 生成器固定 seed
- 不读时钟、不读随机源
